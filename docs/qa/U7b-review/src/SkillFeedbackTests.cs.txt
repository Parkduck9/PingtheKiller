        // Decoy safety (HUD_SPEC 14.0, 6.3): no Survivor skill can produce any Killer-side output, whatever its fields, so a
        // Dummy packet or White noise fake leaves the Killer exactly as a real Signal Survivor does (no skill path at all).
        [Test]
        public void TheKillerHasNoOutputForAnySurvivorSkill()
        {
            foreach (byte skill in new[] { SkillIds.DummyPacket, SkillIds.StealthRelay, SkillIds.TrafficOverload, SkillIds.EmergencyRestore, SkillIds.WhiteNoise })
            foreach (int caster in new[] { Mina, Rio, 0 })
            {
                if (caster == 0 && skill == SkillIds.TrafficOverload) continue; // the one Killer-bound copy: no position, no Id
                var state = new KillerFxState();
                FxRecord r = state.OnEffect(KillerId, caster, skill, 30, 30, 3, 0);
                Assert.That(r.Viewer, Is.EqualTo(FxViewer.None), "skill " + skill + " caster " + caster);
                Assert.That(r.Visual.Shown || r.Sound.Shown || r.Line.Shown, Is.False);
                Assert.That(state.Line(.5).Text, Is.Null);
                Assert.That(state.FovOffset(.5, false), Is.EqualTo(0));
            }
        }

        [Test]
        public void DummyPacketAndWhiteNoiseDecoysAreAbsentFromTheKillerPath()
        {
            // A real Signal Survivor and a fake produce the same Killer-side state: none of either reaches KillerFxState.
            var real = new KillerFxState();
            var fake = new KillerFxState();
            fake.OnEffect(KillerId, Mina, SkillIds.DummyPacket, 30, 30, 3, 0);
            fake.OnEffect(KillerId, Rio, SkillIds.WhiteNoise, 50, 50, 1.5f, 0);
            for (double t = 0; t < 4; t += .25)
            {
                Assert.That(fake.Line(t).Text, Is.EqualTo(real.Line(t).Text));
                Assert.That(fake.FovOffset(t, false), Is.EqualTo(real.FovOffset(t, false)));
                Assert.That(fake.TakeRecoverCue(t), Is.EqualTo(real.TakeRecoverCue(t)));
            }
        }

        [Test]
        public void EffectsLastAtMostTwoSecondsOrTheEffectAndCutsNeverOutliveIt()
        {
            var records = new List<FxRecord>
            {
                SkillFeedback.ForKiller(KillerId, KillerId, SkillIds.PingScanner, 0, 0, 1.5f),
                SkillFeedback.ForKiller(KillerId, KillerId, SkillIds.FirewallBlock, Exit(0).X, Exit(0).Z, 5),
                SkillFeedback.ForSurvivor(Frame(), KillerId, SkillIds.PingScanner, 0, 0, 1.5f),
                SkillFeedback.ForSurvivor(Frame(action: 1), KillerId, SkillIds.FirewallBlock, Exit(0).X, Exit(0).Z, 5),
            };
            foreach (byte skill in new[] { SkillIds.DummyPacket, SkillIds.StealthRelay, SkillIds.TrafficOverload, SkillIds.EmergencyRestore, SkillIds.WhiteNoise })
                records.Add(SkillFeedback.ForSurvivor(Frame(), Own, skill, 40, 40, skill == SkillIds.EmergencyRestore ? 0 : skill == SkillIds.WhiteNoise ? 1.5f : skill == SkillIds.DummyPacket ? 3 : 2));
            float[] durations = { 1.5f, 5, 1.5f, 5, 3, 2, 2, 0, 1.5f };
            for (int i = 0; i < records.Count; i++)
            {
                FxRecord r = records[i];
                if (r.Visual.Shown) Assert.That(r.Visual.Seconds <= 2f || Mathf.Approximately(r.Visual.Seconds, durations[i]), Is.True, "record " + i);
                if (r.Sound.Shown && r.Sound.Cut > 0 && durations[i] > 0) Assert.That(r.Sound.Cut, Is.LessThanOrEqualTo(durations[i] + 1e-4f), "record " + i);
                if (r.Line.Shown) Assert.That(r.Line.Seconds, Is.EqualTo(SkillFeedback.LineSeconds));
            }
            Assert.That(SkillFeedback.Reason("x").Seconds, Is.EqualTo(2f));
            FxVisual relay = records[5].Visual;
            Assert.That(SkillFeedback.DiameterAt(relay, .075), Is.EqualTo(5f).Within(1e-4));
            Assert.That(SkillFeedback.DiameterAt(relay, 1), Is.EqualTo(10f));
            Assert.That(SkillFeedback.DiameterAt(relay, 2), Is.EqualTo(-1f));
        }

        [Test]
        public void PressesConsumedClientSideNeverArmTheRefusedWatch()
        {
            bool watch;
            Assert.That(SkillFeedback.PressReason(Role.Survivor, SkillIds.DummyPacket, PlayerStatus.Lagging, 0, true, out watch), Is.EqualTo("Skills are off while Lagging.")); Assert.That(watch, Is.False);
            Assert.That(SkillFeedback.PressReason(Role.Survivor, SkillIds.DummyPacket, PlayerStatus.Normal, 7, true, out watch), Is.Null); Assert.That(watch, Is.False, "cooldown");
            Assert.That(SkillFeedback.PressReason(Role.Survivor, SkillIds.DummyPacket, PlayerStatus.Normal, null, true, out watch), Is.Null); Assert.That(watch, Is.False, "no SelfState yet");
            Assert.That(SkillFeedback.PressReason(Role.Survivor, SkillIds.DummyPacket, PlayerStatus.Eliminated, 0, true, out watch), Is.Null); Assert.That(watch, Is.False);
            Assert.That(SkillFeedback.PressReason(Role.Survivor, SkillIds.EmergencyRestore, PlayerStatus.Normal, 0, false, out watch), Is.EqualTo("Emergency restore needs a Lagging Survivor next to you.")); Assert.That(watch, Is.False);
            Assert.That(SkillFeedback.PressReason(Role.Killer, SkillIds.FirewallBlock, PlayerStatus.Normal, 0, false, out watch), Is.EqualTo("No Exit switch in progress")); Assert.That(watch, Is.False);
            Assert.That(SkillFeedback.PressReason(Role.Killer, SkillIds.PacketAfterimage, PlayerStatus.Normal, 0, true, out watch), Is.Not.Null); Assert.That(watch, Is.False);
            Assert.That(SkillFeedback.PressReason(Role.Survivor, SkillIds.WhiteNoise, PlayerStatus.Normal, 0, true, out watch), Is.Null); Assert.That(watch, Is.True, "derived Ready");

            var press = new PressWatch();
            press.OnPress(0, false);
            Assert.That(press.Tick(1, 0), Is.False, "a consumed press never infers a refusal");

            Assert.That(press.Tick(.4, 0), Is.False);
            Assert.That(press.Tick(.5, 0), Is.True);
            Assert.That(press.Tick(.6, 0), Is.False, "fires once");
            press.OnPress(0, true);
            Assert.That(press.Tick(.2, 15), Is.False, "the cooldown started: accepted");
            Assert.That(press.Tick(.7, 0), Is.False);
            press.OnPress(0, true); press.OnOwnEffect();
            Assert.That(press.Tick(.7, 0), Is.False, "the own effect arrived");
            var survivor = new SurvivorFxState();
            survivor.Press.OnPress(0, true);
            survivor.OnEffect(Frame(), false, Own, SkillIds.DummyPacket, 40, 40, 3, .1);
            Assert.That(survivor.Press.Armed, Is.False);
        }

        [Test]
        public void SkillFxCreatesNothingAfterWarmUp()
        {
            var host = new GameObject("skill fx test");
            try
            {
                var fx = host.AddComponent<SkillFx>();
                SkillFxArt art = SkillFxArt.Load();
                fx.Warm(art);
                int created = fx.Created, children = host.transform.childCount;
                Assert.That(created, Is.GreaterThan(0));
                var records = new List<FxRecord>
                {
                    SkillFeedback.ForKiller(KillerId, KillerId, SkillIds.PingScanner, 0, 0, 1.5f),
                    SkillFeedback.ForKiller(KillerId, KillerId, SkillIds.FirewallBlock, Exit(0).X, Exit(0).Z, 5),
                    SkillFeedback.ForSurvivor(Frame(), KillerId, SkillIds.PingScanner, 0, 0, 1.5f),
                    SkillFeedback.ForSurvivor(Frame(action: 1), KillerId, SkillIds.FirewallBlock, Exit(0).X, Exit(0).Z, 5),
                    SkillFeedback.ForSurvivor(Frame(mates: Peer(Mina, 10, 10)), Mina, SkillIds.DummyPacket, 10, 10, 3),
                };
                foreach (byte skill in new[] { SkillIds.DummyPacket, SkillIds.StealthRelay, SkillIds.TrafficOverload, SkillIds.EmergencyRestore, SkillIds.WhiteNoise })
                    records.Add(SkillFeedback.ForSurvivor(Frame(), Own, skill, 40, 40, 2));
                double now = 0;
                for (int round = 0; round < 6; round++)
                    foreach (FxRecord r in records) { fx.Play(r, new Vector3(40, 0, 40), now); now += .05; fx.Tick(now); }
                fx.Reduced = true;
                foreach (FxRecord r in records) fx.Play(r, new Vector3(40, 0, 40), now);
                Assert.That(fx.LastKind, Is.Not.EqualTo(FxKind.Hovl), "reduced motion plays no Hovl one-shot");
                Assert.That(fx.Created, Is.EqualTo(created), "no GameObject per effect after warm-up");
                Assert.That(host.transform.childCount, Is.EqualTo(children));
                Assert.That(fx.BusyRings, Is.LessThanOrEqualTo(SkillFx.RingCount));
                fx.Tick(now + 5.1);
                Assert.That(fx.BusyOneShots + fx.BusyRings, Is.EqualTo(0), "everything ends at its time");
                fx.Play(records[5], Vector3.zero, now + 6);
                fx.Clear();
                Assert.That(fx.BusyOneShots + fx.BusyRings + fx.BusyVoices, Is.EqualTo(0));
                if (art != null)
                {
                    Assert.That(art.Marker, Is.Not.Null);
                    Assert.That(art.Marker.GetFloat("_Emission"), Is.LessThanOrEqualTo(.9f), "below the bloom threshold");
                    for (int i = 1; i < art.Clips.Length; i++) Assert.That(art.Clips[i], Is.Not.Null, "clip " + i);
                    for (int i = 1; i < art.Prefabs.Length; i++) Assert.That(art.Prefabs[i], Is.Not.Null, "prefab " + i);
                }
            }
            finally { Object.DestroyImmediate(host); }
        }
